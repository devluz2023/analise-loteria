from itertools import combinations
import pandas as pd
from get_data import df_mega_sena

# 1. Extrair os jogos de forma limpa
colunas_bolas = ['Bola1', 'Bola2', 'Bola3', 'Bola4', 'Bola5', 'Bola6']
sorteios = {}

print("Carregando e organizando os concursos...")
for index, row in df_mega_sena.iterrows():
    concurso = int(row['Concurso'])
    # Guardamos os números como um 'set' (conjunto) para fazer interseção rápida depois
    dezenas = set([int(row[c]) for c in colunas_bolas])
    sorteios[concurso] = dezenas

# Dicionários para guardar o mapeamento de quais concursos repetiram o quê
# Chave: a combinação que repetiu | Valor: lista com os números dos concursos
repeticoes = {6: {}, 5: {}, 4: {}, 3: {}, 2: {}}

print("Analisando a interseção par a par entre todos os concursos... Isso pode levar alguns segundos.")

lista_concursos = sorted(list(sorteios.keys()))
total_concursos = len(lista_concursos)

# 2. Comparar cada concurso com todos os seguintes (evita duplicar A vs B e B vs A)
for i in range(total_concursos):
    c1 = lista_concursos[i]
    jogo1 = sorteios[c1]
    
    for j in range(i + 1, total_concursos):
        c2 = lista_concursos[j]
        jogo2 = sorteios[c2]
        
        # Encontra os números que aparecem nos dois jogos ao mesmo tempo
        numeros_em_comum = jogo1 & jogo2
        quantidade_comum = len(numeros_em_comum)
        
        # Se compartilham de 2 a 6 números, guardamos essa ocorrência de forma exclusiva
        if 2 <= quantidade_comum <= 6:
            # Transformamos em tupla ordenada para servir de chave única no dicionário
            comb_comum = tuple(sorted(list(numeros_em_comum)))
            
            if comb_comum not in repeticoes[quantidade_comum]:
                repeticoes[quantidade_comum][comb_comum] = set([c1, c2])
            else:
                repeticoes[quantidade_comum][comb_comum].add(c1)
                repeticoes[quantidade_comum][comb_comum].add(c2)

# 3. Exibir os Resultados Finais
print("\n" + "=" * 60)
print("ESTATÍSTICAS DE REPETIÇÃO EXCLUSIVA DE JOGOS")
print("=" * 60)

nomes = {6: "SENAS", 5: "QUINAS", 4: "QUADRAS", 3: "TERNOS", 2: "DUQUES"}

for faixa in [6, 5, 4, 3, 2]:
    total_combinacoes_distintas = len(repeticoes[faixa])
    
    print(f"\n▶ {nomes[faixa]} (Compartilhamento exato de {faixa} números):")
    print(f"  Total de grupos de {faixa} números que já se repetiram: {total_combinacoes_distintas}")
    
    if total_combinacoes_distintas > 0:
        # Criar um DataFrame temporário para ordenar e mostrar os exemplos mais frequentes
        dados_faixa = []
        for comb, concs in repeticoes[faixa].items():
            dados_faixa.append({
                'Combinação': comb,
                'Vezes': len(concs),
                'Concursos': sorted(list(concs))
            })
        
        df_faixa = pd.DataFrame(dados_faixa).sort_values(by='Vezes', ascending=False)
        print("  Top 3 combinações mais repetidas nessa faixa:")
        print(df_faixa.head(3).to_string(index=False))
    else:
        print("  Nenhuma ocorrência registrada até hoje.")

print("\n" + "=" * 60)

# 3. MOTOR DE VALIDAÇÃO FIEL ÀS PROBABILIDADES DAS COLUNAS (1 A 108)

def jogo_e_valido_estritamente(novo_jogo):
    novo_jogo_tuple = tuple(novo_jogo)
    
    # ---------------------------------------------------------------------
    # BARREIRA 1: ADERÊNCIA À CURVA MATRICIAL DE PROBABILIDADE DE JEJUM
    # ---------------------------------------------------------------------
    probabilidades_das_6_dezenas = []
    
    for num in novo_jogo:
        # Se a dezena sair hoje, o ciclo fecha no tamanho atual dela + 1
        jejum_futuro = atraso_atual_por_dezena[num] + 1
        
        # Puxa do nosso dicionário qual é a probabilidade geral de um jejum fechar com esse tamanho
        chance_desse_jejum = probabilidade_tamanho_jejum.get(jejum_futuro, 0)
        
        probabilidades_das_6_dezenas.append(chance_desse_jejum)
        
    # Critério de Rejeição 1: Se o jogo contiver qualquer dezena cujo tamanho de
    # jejum futuro seja uma anomalia absoluta (chance menor que 0.05% ou 0.0005)
    # permitindo apenas se for jejum ultra baixo (1 ou 2) que são zonas de repetição rápida
    for p, num in zip(probabilidades_das_6_dezenas, novo_jogo):
        if p < 0.0005 and (atraso_atual_por_dezena[num] + 1) > 2:
            return False # Descarta na hora por ser um tamanho de jejum improvável de fechar hoje

    # Critério de Rejeição 2: O equilíbrio coletivo das probabilidades.
    # Em média, um jogo normal sorteado é composto por dezenas que flutuam em posições
    # confortáveis da curva. Se o produto ou a média for perigosamente baixa, o jogo é descartado.
    if np.mean(probabilidades_das_6_dezenas) < 0.015:
        return False

    # ---------------------------------------------------------------------
    # BARREIRA 2: FILTRO DE REPETIÇÃO ESTOCÁSTICA (QUADRAS/TERNOS)
    # ---------------------------------------------------------------------
    if novo_jogo_tuple in senas_historicas:
        return False
        
    for quina in combinations(novo_jogo, 5):
        if quina in quinas_historicas and random.random() > probabilidades_repeticao[5]:
            return False
                
    contagem_quadras = sum(1 for quadra in combinations(novo_jogo, 4) if quadra in quadras_historicas)
    if contagem_quadras > 0 and random.random() > probabilidades_repeticao[4]:
        return False

    contagem_ternos = sum(1 for terno in combinations(novo_jogo, 3) if terno in ternos_historicas)
    if contagem_ternos < 1 or contagem_ternos > 3:
        return False
        
    return True

print("Carregando dados e construindo Autoencoder para a Matriz de Bayes...")
colunas_bolas = ['Bola1', 'Bola2', 'Bola3', 'Bola4', 'Bola5', 'Bola6']
df_cronologico = df_mega_sena.sort_values(by='Concurso').copy()

# Prepara os dados para o Autoencoder (Instâncias de jejum histórico)
matriz_jejuns_bruta = df_ausencia[colunas_jejum].values.astype(np.float32)
scaler_ae = MinMaxScaler(feature_range=(0, 1))
matriz_jejuns_norm = scaler_ae.fit_transform(matriz_jejuns_bruta)

# Arquitetura do Autoencoder
dim_entrada = matriz_jejuns_norm.shape[1] # Qtd de colunas de jejum (ex: 108)
dim_latente = 12                          # Gargalo: Comprime para 12 dimensões essenciais

input_layer = Input(shape=(dim_entrada,))
encoded = Dense(32, activation='relu')(input_layer)
bottleneck = Dense(dim_latente, activation='relu', name='espaco_latente')(encoded)
decoded = Dense(32, activation='relu')(bottleneck)
output_layer = Dense(dim_entrada, activation='sigmoid')(decoded)

autoencoder = Model(input_layer, output_layer)
encoder = Model(input_layer, bottleneck) # Modelo cortado que extrai os recursos comprimidos

autoencoder.compile(optimizer='adam', loss='mean_squared_error')
# Treina o Autoencoder para reconstruir a si mesmo (Self-Supervised)
autoencoder.fit(matriz_jejuns_norm, matriz_jejuns_norm, epochs=40, batch_size=32, verbose=0)

# Extrai os scores latentes e mapeia de volta um vetor de peso para cada tamanho de jejum
recursos_latentes = encoder.predict(matriz_jejuns_norm, verbose=0)
score_peso_jejum = np.mean(recursos_latentes, axis=0) # Perfil latente médio

# Reconstrói a estrutura a priori para o histórico
distribuicao_global_jejuns = {int(col): 0 for col in colunas_jejum}

total_sequencias = sum(distribuicao_global_jejuns.values())
prob_prior_jejum = {t: (qtd / total_sequencias) for t, qtd in distribuicao_global_jejuns.items()}

historico_atrasos = []

# 2. OPTIMIZADOR NÃO LINEAR COM ENGENHARIA LATENTE DO AUTOENCODER
# =============================================================
def resolver_nlp_com_recursos_autoencoder(scores_dezenas, atrasos_atuais, params_nlp):
    """
    Usa SLSQP para otimizar a escolha das dezenas aplicando pesos ajustados
    pelas features compactadas do Autoencoder.
    """
    scores = np.array(scores_dezenas)
    
    def funcao_objetivo(x):
        x_safe = np.clip(x, 1e-10, 1.0)
        # Termo Não Linear: Entropia Ponderada pelo perfil latente do Autoencoder
        # Força o espalhamento de dezenas nas dimensões que o Autoencoder identificou como estruturais
        entropia_latente = -np.sum(x_safe * np.log(x_safe)) * np.mean(score_peso_jejum)
        return -(np.sum(scores * x) + params_nlp['peso_entropia'] * entropia_latente)

    def restricao_soma(x):
        return np.sum(x) - 6.0

    def restricao_frias(x):
        frias_idx = np.where(atrasos_atuais > params_nlp['corte_jejum_longo'])[0]
        return params_nlp['max_dezenas_frias'] - np.sum(x[frias_idx])

    def restricao_quentes(x):
        quentes_idx = np.where(atrasos_atuais <= 4)[0]
        return np.sum(x[quentes_idx]) - params_nlp['min_dezenas_quentes']

    constraints = [
        {'type': 'eq', 'fun': restricao_soma},
        {'type': 'ineq', 'fun': restricao_frias},
        {'type': 'ineq', 'fun': restricao_quentes}
    ]
    
    bounds = [(0.0, 1.0) for _ in range(60)]
    x0 = np.full(60, 0.1)
    
    res = minimize(funcao_objetivo, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 80, 'disp': False})
    
    if res.success:
        return {d+1: max(0.0, val) for d, val in enumerate(res.x)}
    else:
        return {d+1: 0.1 for d in range(60)}


def amostrar_jogos_da_curva_latente(pesos_nlp, n_jogos=5):
    jogos_gerados = []
    dezenas = list(range(1, 61))
    valores_pesos = np.array([pesos_nlp[d] for d in dezenas])
    valores_pesos = np.where(valores_pesos < 1e-4, 1e-5, valores_pesos)
    
    tentativas = 0
    while len(jogos_gerados) < n_jogos and tentativas < 2000:
        tentativas += 1
        probs_roleta = valores_pesos / np.sum(valores_pesos)
        palpite = sorted(np.random.choice(dezenas, size=6, replace=False, p=probs_roleta))
        
        if sum(1 for n in palpite if n % 2 == 0) in [2, 3, 4]:
            if palpite not in jogos_gerados:
                jogos_gerados.append([int(n) for n in palpite])
                
    return jogos_gerados

# 3. FUNÇÃO OBJETIVO DO OPTUNA

def objective(trial):
    window_size = trial.suggest_int('window_size', 10, 16)
    peso_lstm = trial.suggest_float('peso_lstm', 0.4, 0.7)
    peso_bayes = 1.0 - peso_lstm
    
    params_nlp = {
        'peso_entropia': trial.suggest_float('peso_entropia', 0.02, 0.20),
        'corte_jejum_longo': trial.suggest_int('corte_jejum_longo', 14, 24),
        'max_dezenas_frias': trial.suggest_float('max_dezenas_frias', 1.0, 2.0),
        'min_dezenas_quentes': trial.suggest_float('min_dezenas_quentes', 2.0, 3.5)
    }
    
    X_all, y_all, gabarito_all = [], [], []
    for i in range(len(dados_norm) - window_size):
        X_all.append(dados_norm[i:(i + window_size), :])
        y_all.append(dados_norm[i + window_size, :])
        gabarito_all.append(gabarito_sorteios[i + window_size])

    X_all, y_all = np.array(X_all), np.array(y_all)
    train_size = int(len(X_all) * 0.8)
    X_train, X_test = X_all[:train_size], X_all[train_size:]
    y_train = y_all[:train_size]
    gabarito_teste = gabarito_all[train_size:]
    
    model = Sequential([
        LSTM(64, return_sequences=False, input_shape=(window_size, 60)),
        Dropout(0.2),
        Dense(60, activation='sigmoid')
    ])
    model.compile(optimizer='adam', loss='mean_squared_error')
    model.fit(X_train, y_train, batch_size=64, epochs=8, verbose=0)
    
    previsoes_norm = model.predict(X_test[-60:], verbose=0)
    previsoes_lstm = scaler_lstm.inverse_transform(previsoes_norm)
    gabarito_recortado = gabarito_teste[-60:]
    atrasos_no_teste = matriz_atrasos[train_size + window_size - 1 : -1][-60:]
    
    score_fim = 0
    for i in range(60):
        pred_lstm = previsoes_lstm[i]
        atraso_concurso_atual = atrasos_no_teste[i]
        resultado_real = gabarito_recortado[i]
        
        scores_dezenas = []
        for idx in range(60):
            score_lstm_norm = 1 / (1 + pred_lstm[idx])
            jejum_futuro = atraso_concurso_atual[idx] + 1
            score_bayes_norm = prob_prior_jejum.get(jejum_futuro, 0.0001)
            scores_dezenas.append((peso_lstm * score_lstm_norm) + (peso_bayes * score_bayes_norm))
            
        pesos_nlp = resolver_nlp_com_recursos_autoencoder(scores_dezenas, atraso_concurso_atual, params_nlp)
        jogos_simulados = amostrar_jogos_da_curva_latente(pesos_nlp, n_jogos=5)
                
        for jogo in jogos_simulados:
            acertos = len(set(jogo) & resultado_real)
            if acertos == 4: score_fim += 1000
            elif acertos == 5: score_fim += 50000
            
    return score_fim

    caminharem sob a influência das probabilidades da rede neural LSTM.
    """
    dezenas = list(range(1, 61))
    frequencia_combinacoes_validas = {}
    
    prob_markov_quebra = []
    for idx in range(60):
        jejum_hoje = atrasos_atuais[idx]
        p_zero_markov = matriz_transicao_markov.get(jejum_hoje, {0: 0.1})[0]
        prob_markov_quebra.append(p_zero_markov)
        
    prob_markov_quebra = np.array(prob_markov_quebra)
    score_ia_temporal = 1 / (1 + pred_lstm)
    
    propensao_sorteio = prob_markov_quebra * score_ia_temporal
    probs_roleta = propensao_sorteio / np.sum(propensao_sorteio)
    
    for _ in range(n_simulacoes_mcmc):
        cenario_sorteio = sorted(np.random.choice(dezenas, size=6, replace=False, p=probs_roleta))
        
        # CORRIGIDO: Corrigida a chamada da função para bater com o import do seu gerador.py
        if jogo_e_valido_estritamente(cenario_sorteio):
            tuple_jogo = tuple(cenario_sorteio)
            if tuple_jogo in frequencia_combinacoes_validas:
                frequencia_combinacoes_validas[tuple_jogo] += 1
            else:
                frequencia_combinacoes_validas[tuple_jogo] = 1
                
    jogos_ordenados = sorted(frequencia_combinacoes_validas.items(), key=lambda x: x[1], reverse=True)
    return [list(item[0]) for item in jogos_ordenados[:5]]

# 2. MOTOR DE SIMULAÇÃO DE MONTE CARLO (MCMC ENGINE)
# =============================================================
def simular_monte_carlo_proximo_concurso(scores_dezenas, atrasos_atuais, n_simulacoes=5000):
    """
    Simula milhares de cenários possíveis para o sorteio usando amostragem
    estocástica baseada nas notas da IA e validando nas restrições de Bayes.
    """
    dezenas = list(range(1, 61))
    scores_array = np.array(scores_dezenas)
    
    # Aplica Softmax para transformar os scores em uma distribuição de probabilidade estrita para a roleta
    exp_scores = np.exp(scores_array / np.max(scores_array)) # Temperatura de amostragem
    probs_roleta = exp_scores / np.sum(exp_scores)
    
    # Dicionário para computar a frequência de aparição de cada jogo válido simulado
    historico_cenarios_validos = {}
    
    for _ in range(n_simulacoes):
        # Monte Carlo: Sorteia um cenário candidado com base nas probabilidades da IA
        cenario_palpite = sorted(np.random.choice(dezenas, size=6, replace=False, p=probs_roleta))
        
        # Filtro de Rejeição de Monte Carlo: O cenário só é aceito se passar no seu gerador estrito
        if jogo_e_valido_estritamente(cenario_palpite):
            # Filtro adicional de Bayes: Evitar acumular dezenas extremamente frias no mesmo cenário
            frias = sum(1 for d in cenario_palpite if atrasos_atuais[d-1] > 18)
            if frias <= 2:
                tuple_jogo = tuple(cenario_palpite)
                if tuple_jogo in historico_cenarios_validos:
                    historico_cenarios_validos[tuple_jogo] += 1
                else:
                    historico_cenarios_validos[tuple_jogo] = 1
                    
    # Ordena as combinações que mais vezes convergiram e sobreviveram em todos os universos simulados
    jogos_mais_frequentes = sorted(historico_cenarios_validos.items(), key=lambda x: x[1], reverse=True)
    
    # Retorna o top 5 de jogos mais estáveis encontrados por Monte Carlo
    return [list(item[0]) for item in jogos_mais_frequentes[:5]]

# 2. OTIMIZADOR NÃO LINEAR (NLP VIA SLSQP)

def resolver_problema_nao_linear(scores_dezenas, atrasos_atuais, params_nlp):
    """
    Usa Otimização Não Linear Restrita para maximizar a aderência combinada
    enquanto gerencia a entropia (grau de desordem/surpresa) do bilhete.
    """
    probs_bayes = np.array([prob_prior_jejum.get(atrasos_atuais[d] + 1, 0.0001) for d in range(60)])
    scores = np.array(scores_dezenas)
    
    # 2.1 FUNÇÃO OBJETIVO NÃO LINEAR: Maximização com penalidade de entropia
    # Queremos maximizar o score, o que equivale a minimizar o negativo dele
    def funcao_objetivo(x):
        # Evita log de zero adicionando um epsilon minúsculo
        x_safe = np.clip(x, 1e-10, 1.0)
        # Termo Não Linear: Entropia de Shannon (mede a dispersão e incerteza da distribuição)
        entropia = -np.sum(x_safe * np.log(x_safe))
        
        # O retorno combina os ganhos de acerto da IA subtraindo ou somando a desordem (regulação não linear)
        return -(np.sum(scores * x) + params_nlp['peso_entropia'] * entropia)

    # 2.2 RESTRIÇÕES (CONSTRAINTS)
    # Restrição Linear Básica: A soma dos pesos contínuos precisa dar exatamente 6.0
    def restricao_soma(x):
        return np.sum(x) - 6.0

    # Restrição Não Linear de Bayes: O produto/soma geométrica das dezenas frias precisa respeitar o teto
    def restricao_frias(x):
        frias_idx = np.where(atrasos_atuais > params_nlp['corte_jejum_longo'])[0]
        return params_nlp['max_dezenas_frias'] - np.sum(x[frias_idx])

    def restricao_quentes(x):
        quentes_idx = np.where(atrasos_atuais <= 4)[0]
        return np.sum(x[quentes_idx]) - params_nlp['min_dezenas_quentes']

    # Encapsula as restrições no formato exigido pelo Scipy (type 'eq' para igualdade, 'ineq' para >= 0)
    constraints = [
        {'type': 'eq', 'fun': restricao_soma},
        {'type': 'ineq', 'fun': restricao_frias},
        {'type': 'ineq', 'fun': restricao_quentes}
    ]
    
    # Limites das variáveis: cada peso x_d deve flutuar estritamente entre 0.0 e 1.0
    bounds = [(0.0, 1.0) for _ in range(60)]
    
    # Chute inicial: distribui o peso igualmente (6 dezenas / 60 posições = 0.1 para cada)
    x0 = np.full(60, 0.1)
    
    # Executa o solver NLP SLSQP
    res = minimize(funcao_objetivo, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100, 'disp': False})
    
    if res.success:
        return {d+1: max(0.0, val) for d, val in enumerate(res.x)}
    else:
        # Fallback seguro caso o gradiente não convirja
        return {d+1: 0.1 for d in range(60)}


def amostrar_jogos_da_curva_nao_linear(pesos_nlp, n_jogos=5):
    """ Converte a curva de densidade não linear em jogos de 6 dezenas """
    jogos_gerados = []
    dezenas = list(range(1, 61))
    valores_pesos = np.array([pesos_nlp[d] for d in dezenas])
    valores_pesos = np.where(valores_pesos < 1e-4, 1e-5, valores_pesos)
    
    tentativas = 0
    while len(jogos_gerados) < n_jogos and tentativas < 2000:
        tentativas += 1
        probs_roleta = valores_pesos / np.sum(valores_pesos)
        palpite = sorted(np.random.choice(dezenas, size=6, replace=False, p=probs_roleta))
        
        if sum(1 for n in palpite if n % 2 == 0) in [2, 3, 4]:
            if palpite not in jogos_gerados:
                jogos_gerados.append([int(n) for n in palpite])
                
    return jogos_gerados

# =============================================================
# 2. MOTOR DE PESQUISA OPERACIONAL (SOLVER MIP - PuLP)
# =============================================================
def solver_gerar_jogos_com_restricoes(scores_dezenas, n_jogos=5, jogos_proibidos=None):
    """
    Usa programação linear inteira (MIP) para construir bilhetes perfeitos
    que atendem estritamente a todas as regras geométricas e estocásticas.
    """
    jogos_otimizados = []
    if jogos_proibidos is None:
        jogos_proibidos = []
        
    # Mapeamento estrito de quadrantes da Mega-Sena
    q1 = [n for n in range(1, 31) if n % 10 in [1, 2, 3, 4, 5]]
    q2 = [n for n in range(1, 31) if n % 10 in [6, 7, 8, 9, 0]]
    q3 = [n for n in range(31, 61) if n % 10 in [1, 2, 3, 4, 5]]
    q4 = [n for n in range(31, 61) if n % 10 in [6, 7, 8, 9, 0]]

    for rodada in range(n_jogos):
        # Inicializa o problema de maximização
        prob = LpProblem(f"Otimizacao_Mega_Sena_Rodada_{rodada}", LpMaximize)
        
        # Variáveis de Decisão: x[i] é binária (1 se a dezena i for escolhida, 0 se não)
        x = LpVariable.dicts("dezena", range(1, 61), cat="Binary")
        
        # Função Objetivo: Maximizar o score combinado (IA + Bayes) das dezenas escolhidas
        prob += lpSum(scores_dezenas[d-1] * x[d] for d in range(1, 61))
        
        # --- RESTRICÕES RÍGIDAS DE JOGO VÁLIDO ---
        # 1. O jogo precisa ter exatamente 6 dezenas
        prob += lpSum(x[d] for d in range(1, 61)) == 6
        
        # 2. Paridade Estrita: Exatamente 3 pares e 3 ímpares
        prob += lpSum(x[d] for d in range(1, 61) if d % 2 == 0) == 3
        prob += lpSum(x[d] for d in range(1, 61) if d % 2 != 0) == 3
        
        # 3. Equilíbrio Espacial de Quadrantes: No máximo 2 dezenas por quadrante
        prob += lpSum(x[d] for d in q1) <= 2
        prob += lpSum(x[d] for d in q2) <= 2
        prob += lpSum(x[d] for d in q3) <= 2
        prob += lpSum(x[d] for d in q4) <= 2
        
        # 4. Restrição de Finais repetidos: No máximo 2 números com o mesmo final (ex: no máximo dois finais '4')
        for f in range(10):
            dezenas_final_f = [d for d in range(1, 61) if d % 10 == f or (f == 0 and d % 10 == 0)]
            prob += lpSum(x[d] for d in dezenas_final_f) <= 2

        # 5. Restrição de Repetição Inter-Jogos (Garantir diversidade entre os 5 bilhetes gerados)
        for jogo_existente in jogos_otimizados + jogos_proibidos:
            # Não permite que o novo jogo compartilhe mais do que 3 dezenas com os anteriores
            prob += lpSum(x[d] for d in jogo_existente) <= 3
            
        # Executa o solver silenciosamente
        prob.solve(PULP_CBC_CMD(msg=False))
        
        if LpStatus[prob.status] == "Optimal":
            jogo_resolvido = [d for d in range(1, 61) if x[d].varValue == 1]
            jogos_otimizados.append(sorted(jogo_resolvido))
        else:
            # Se o espaço fechar demais por restrição, afrouxa o teto de repetição para não travar
            break
            
    return jogos_otimizados

def objective(trial):
    window_size = trial.suggest_int('window_size', 10, 18)
    peso_lstm = trial.suggest_float('peso_lstm', 0.3, 0.7)
    peso_bayes = 1.0 - peso_lstm
    
    X_all, y_all, gabarito_all = [], [], []
    for i in range(len(dados_norm) - window_size):
        X_all.append(dados_norm[i:(i + window_size), :])
        y_all.append(dados_norm[i + window_size, :])
        gabarito_all.append(gabarito_sorteios[i + window_size])

    X_all, y_all = np.array(X_all), np.array(y_all)
    train_size = int(len(X_all) * 0.8)
    X_train, X_test = X_all[:train_size], X_all[train_size:]
    y_train = y_all[:train_size]
    gabarito_teste = gabarito_all[train_size:]
    
    model = Sequential([
        LSTM(64, return_sequences=False, input_shape=(window_size, 60)),
        Dropout(0.2),
        Dense(60, activation='sigmoid')
    ])
    model.compile(optimizer='adam', loss='mean_squared_error')
    model.fit(X_train, y_train, batch_size=64, epochs=8, verbose=0)
    
    previsoes_norm = model.predict(X_test[-60:], verbose=0)
    previsoes_lstm = scaler.inverse_transform(previsoes_norm)
    gabarito_recortado = gabarito_teste[-60:]
    atrasos_no_teste = matriz_atrasos[train_size + window_size - 1 : -1][-60:]
    
    score_fim = 0
    for i in range(60):
        pred_lstm = previsoes_lstm[i]
        atraso_concurso_atual = atrasos_no_teste[i]
        resultado_real = gabarito_recortado[i]
        
        # Calcula vetor de utilidade de 60 posições para alimentar o solver
        scores_dezenas = []
        for idx in range(60):
            score_lstm_norm = 1 / (1 + pred_lstm[idx])
            jejum_futuro = atraso_concurso_atual[idx] + 1
            score_bayes_norm = prob_prior_jejum.get(jejum_futuro, 0.0001)
            scores_dezenas.append((peso_lstm * score_lstm_norm) + (peso_bayes * score_bayes_norm))
            
        # O SOLVER ENTRA EM AÇÃO AQUI: Monta matematicamente os jogos ótimos
        jogos_simulados = solver_gerar_jogos_com_restricoes(scores_dezenas, n_jogos=5)
                
        for jogo in jogos_simulados:
            acertos = len(set(jogo) & resultado_real)
            if acertos == 4: score_fim += 1000   # Foco cirúrgico em Quadras
            elif acertos == 5: score_fim += 50000
            
    return score_fim

# 2. SOLVER MIP COM RESTRIÇÕES DA MATRIZ DE PROBABILIDADE
# =============================================================
def solver_gerar_jogos_bayes_constraints(scores_dezenas, atrasos_atuais, config_po, n_jogos=5, jogos_proibidos=None):
    """
    Usa Programação Inteira para construir jogos baseando-se estritamente
    nas restrições probabilísticas da matriz de ausência.
    """
    jogos_otimizados = []
    if jogos_proibidos is None:
        jogos_proibidos = []

    # Mapeamento de probabilidade individual para cada dezena baseada no seu atraso hoje
    probs_dezenas_atuais = {
        d: prob_prior_jejum.get(atrasos_atuais[d-1] + 1, 0.0001) for d in range(1, 61)
    }

    for rodada in range(n_jogos):
        prob = LpProblem(f"Otimizacao_Bayes_MIP_{rodada}", LpMaximize)
        x = LpVariable.dicts("dezena", range(1, 61), cat="Binary")
        
        # Objetivo: Maximizar o score combinado do Ensemble (IA + Bayes)
        prob += lpSum(scores_dezenas[d-1] * x[d] for d in range(1, 61))
        
        # --- RESTRICÕES ESTRUTURAIS BÁSICAS ---
        prob += lpSum(x[d] for d in range(1, 61)) == 6
        prob += lpSum(x[d] for d in range(1, 61) if d % 2 == 0) == 3
        prob += lpSum(x[d] for d in range(1, 61) if d % 2 != 0) == 3
        
        # --- RESTRICÕES DIRETAS DA MATRIZ DE PROBABILIDADE (BAYES) ---
        # 1. Restrição de dezenas "Frias" (Com jejum longo):
        # Obriga o solver a escolher no máximo X dezenas que estejam com atraso maior que o corte estipulado
        prob += lpSum(x[d] for d in range(1, 61) if atrasos_atuais[d-1] > config_po['corte_jejum_longo']) <= config_po['max_dezenas_frias']
        
        # 2. Restrição de dezenas "Quentes" (Com jejum curto):
        # Garante que o jogo tenha pelo menos Y dezenas com alto índice de fechamento de ciclo (atraso curto)
        prob += lpSum(x[d] for d in range(1, 61) if atrasos_atuais[d-1] <= 4) >= config_po['min_dezenas_quentes']
        
        # 3. Janela de Probabilidade Coletiva:
        # A soma das probabilidades de Bayes do bilhete não pode ser absurdamente baixa (jogo impossível) nem alta demais (jogo óbvio)
        prob += lpSum(probs_dezenas_atuais[d] * x[d] for d in range(1, 61)) >= config_po['min_soma_bayes']
        prob += lpSum(probs_dezenas_atuais[d] * x[d] for d in range(1, 61)) <= config_po['max_soma_bayes']

        # 4. Evitar repetição pesada com bilhetes gerados anteriormente nas rodadas
        for jogo_existente in jogos_otimizados + jogos_proibidos:
            prob += lpSum(x[d] for d in jogo_existente) <= 3
            
        # Executa o solver
        prob.solve(PULP_CBC_CMD(msg=False))
        
        if LpStatus[prob.status] == "Optimal":
            jogo_resolvido = [d for d in range(1, 61) if x[d].varValue == 1]
            jogos_otimizados.append(sorted(jogo_resolvido))
        else:
            # Se o espaço de restrição probabilística for violado, quebra e retorna o que conseguiu
            break
            
    return jogos_otimizados

def resolver_problema_linear_puro(scores_dezenas, atrasos_atuais, config_po):
    """
    Resolve a distribuição de pesos usando Programação Linear Contínua (Simplex).
    Retorna o vetor de pesos ótimos fracionários para cada dezena.
    """
    probs_dezenas_atuais = {
        d: prob_prior_jejum.get(atrasos_atuais[d-1] + 1, 0.0001) for d in range(1, 61)
    }

    prob = LpProblem("Maximizacao_Linear_Pura_Simplex", LpMaximize)
    
    # VARIÁVEIS CONTÍNUAS: Substitui as binárias (MIP) por floats entre 0.0 e 1.0
    x = LpVariable.dicts("peso_dezena", range(1, 61), lowBound=0.0, upBound=1.0, cat="Continuous")
    
    # Função Objetivo: Maximizar o valor esperado do bilhete contínuo
    prob += lpSum(scores_dezenas[d-1] * x[d] for d in range(1, 61))
    
    # --- RESTRIÇÕES LINEARES DA MATRIZ ---
    # A soma dos pesos distribuídos na matriz de 60 dezenas deve ser exatamente igual a 6
    prob += lpSum(x[d] for d in range(1, 61)) == 6.0
    
    # Restrição de dezenas frias e quentes operando sob frações contínuas
    prob += lpSum(x[d] for d in range(1, 61) if atrasos_atuais[d-1] > config_po['corte_jejum_longo']) <= config_po['max_dezenas_frias']
    prob += lpSum(x[d] for d in range(1, 61) if atrasos_atuais[d-1] <= 4) >= config_po['min_dezenas_quentes']
    
    # Janela de envelope de Bayes contínuo
    prob += lpSum(probs_dezenas_atuais[d] * x[d] for d in range(1, 61)) >= config_po['min_soma_bayes']
    prob += lpSum(probs_dezenas_atuais[d] * x[d] for d in range(1, 61)) <= config_po['max_soma_bayes']
    
    prob.solve(PULP_CBC_CMD(msg=False))
    
    # Se o simplex convergir para o ótimo, extrai o dicionário de pesos
    if LpStatus[prob.status] == "Optimal":
        return {d: max(0.0, x[d].varValue) for d in range(1, 61)}
    else:
        # Fallback caso os limites de Bayes fiquem travados demais
        return {d: 1.0 if d in sorted(random.sample(range(1, 61), 15)) else 0.0 for d in range(1, 61)}


def amostrar_jogos_da_distribuicao_linear(pesos_continuos, n_jogos=5):
    """
    Usa amostragem probabilística ponderada para converter a solução fracionária
    do Simplex em combinações reais de 6 números.
    """
    jogos_gerados = []
    dezenas = list(range(1, 61))
    
    # Normaliza e adiciona um ruído suave para garantir diversidade entre os bilhetes
    valores_pesos = np.array([pesos_continuos[d] for d in dezenas])
    valores_pesos = np.where(valores_pesos < 0.01, 0.0001, valores_pesos) # Evita probabilidade zero absoluta
    
    tentativas = 0
    while len(jogos_gerados) < n_jogos and tentativas < 2000:
        tentativas += 1
        # Sorteia 6 dezenas usando os pesos do solver contínuo como pesos de probabilidade
        probs_roleta = valores_pesos / np.sum(valores_pesos)
        palpite = sorted(np.random.choice(dezenas, size=6, replace=False, p=probs_roleta))
        
        # Filtros básicos de higienização geométrica pós-amostragem
        if sum(1 for n in palpite if n % 2 == 0) in [2, 3, 4]:
            if palpite not in jogos_gerados:
                jogos_gerados.append([int(n) for n in palpite])
                
    return jogos_gerados

import pandas as pd
from math import comb
from get_data import df_mega_sena

# 1. Identificar as colunas das dezenas (certifique-se que os nomes estão corretos no seu Excel)
# Com base no seu comando, as colunas geralmente são 'Bola1', 'Bola2', etc.
colunas_bolas = ['Bola1', 'Bola2', 'Bola3', 'Bola4', 'Bola5', 'Bola6']

def encontrar_posicao_itertools(linha):
    """
    Calcula a posição exata que uma combinação ocuparia no
    gerador itertools.combinations(range(1, 61), 6)
    """
    # Extrair os números da linha e garantir que estão ordenados
    dezenas = sorted([int(n) for n in linha[colunas_bolas]])

    n = 60
    k = 6
    posicao = 0
    valor_anterior = 1

    for i, valor in enumerate(dezenas):
        # Somamos as combinações que foram "puladas"
        for v in range(valor_anterior, valor):
            posicao += comb(n - v, k - 1 - i)
        valor_anterior = valor + 1

    return posicao + 1

# 2. Criar a nova coluna no DataFrame
print("Calculando o grupo/índice de cada sorteio...")
df_mega_sena['Indice_Combinacao'] = df_mega_sena.apply(encontrar_posicao_itertools, axis=1)

# 3. Exibir o resultado final com a nova coluna
print(df_mega_sena[['Concurso', 'Indice_Combinacao'] + colunas_bolas].head())

                if atraso_atual[dezena] > 0:
                    all_absence_streaks_per_dezena[dezena].append(atraso_atual[dezena])
                    atraso_atual[dezena] = 0
                
                # Controle de saídas consecutivas (saiu jogo seguido)
                if saiu_no_anterior[dezena]:
                    streak_consecutivo_atual[dezena] += 1
                else:
                    streak_consecutivo_atual[dezena] = 1
                
                if streak_consecutivo_atual[dezena] > max_consecutivo_sorteios[dezena]:
                    max_consecutivo_sorteios[dezena] = streak_consecutivo_atual[dezena]
                    
                saiu_no_anterior[dezena] = True
            else:
                # Se não saiu, acumula o jejum atual
                atraso_atual[dezena] += 1
                streak_consecutivo_atual[dezena] = 0
                saiu_no_anterior[dezena] = False

    # Descobrir qual foi o maior jejum que aconteceu na história com qualquer dezena
    max_observed_streak = 0
    for dezena, streaks in all_absence_streaks_per_dezena.items():
        if streaks:
            max_observed_streak = max(max_observed_streak, max(streaks))

    # 3. Construção das Colunas Estatísticas e de Resumo
    rows_data = []

    # Probabilidade teórica de uma dezena NÃO sair em 1 concurso: 54/60 = 0.90
    prob_teorica_nao_sair = 54.0 / 60.0 

    for dezena in range(1, 61):
        streaks = all_absence_streaks_per_dezena[dezena]
        streak_counts = Counter(streaks)
        
        # Contagens de sequências
        total_sequences_any = len(streaks)
        total_sequences_greater_1 = sum(1 for s in streaks if s > 1)
        
        total_absence_draws = sum(streaks)
        average_streak_len = total_absence_draws / total_sequences_any if total_sequences_any > 0 else 0
        
        # Cálculos de Probabilidade Empírica (Baseada no Histórico Real)
        # Total de vezes que a dezena NÃO saiu na história inteira dividido pelo total de concursos
        vezes_nao_sorteada = total_concursos - df_mega_sena[colunas_bolas].isin([dezena]).any(axis=1).sum()
        prob_nao_sair_1_concurso = vezes_nao_sorteada / total_concursos
        prob_nao_sair_max_concursos = prob_nao_sair_1_concurso ** max_observed_streak
        
        # Frequência Média Esperada e Desvio Padrão Teórico
        frequencia_media_esperada = total_concursos * (6.0 / 60.0) # Esperado de saídas = 10% dos concursos
        saidas_reais = total_concursos - vezes_nao_sorteada
        desvio_padrao_observado = saidas_reais - frequencia_media_esperada

        # Monta a linha inicial com os cabeçalhos estatísticos idênticos ao seu modelo
        row_entry = {
            'Dezena': dezena,
            'FrequenciaMedia_Esperada': round(frequencia_media_esperada, 1),
            'Desvio': round(desvio_padrao_observado, 1),
            'Prob_Nao_Sair_Empirica_1_Concurso': round(prob_nao_sair_1_concurso, 4),
            f'Prob_Nao_Sair_Empirica_{max_observed_streak}_Concursos': f"{prob_nao_sair_max_concursos:.6e}",
            'Máximo Consecutivo': max_consecutivo_sorteios[dezena],
            'Total de Sequências (qualquer tamanho)': total_sequences_any,
            'Total de Sequências (> 1)': total_sequences_greater_1,
            'Máximo de Sorteios Sem Sair': max(streaks) if streaks else 0,
            'Ausencia_Actual_Concursos': atraso_atual[dezena]
        }
        
        # Preenche a contagem de vezes para cada tamanho de jejum (Coluna 1 até o Máximo Histórico)
        for streak_len in range(1, max_observed_streak + 1):
            row_entry[str(streak_len)] = streak_counts.get(streak_len, 0)
            
        # Colunas finais de consolidação
        row_entry['Total_Streaks_Observed'] = total_sequences_any
        row_entry['Total_Absence_Draws'] = total_absence_draws
        row_entry['Average_Streak_Length'] = round(average_streak_len, 4)
        
        rows_data.append(row_entry)

    # 4. Organizar Ordem das Colunas na Tabela Final
    colunas_estatisticas = [
        'Dezena', 'FrequenciaMedia_Esperada', 'Desvio', 
        'Prob_Nao_Sair_Empirica_1_Concurso', f'Prob_Nao_Sair_Empirica_{max_observed_streak}_Concursos',
        'Máximo Consecutivo', 'Total de Sequências (qualquer tamanho)', 'Total de Sequências (> 1)', 
        'Máximo de Sorteios Sem Sair', 'Ausencia_Actual_Concursos'
    ]
    colunas_dinamicas_jejum = [str(i) for i in range(1, max_observed_streak + 1)]
    colunas_resumo_finais = ['Total_Streaks_Observed', 'Total_Absence_Draws', 'Average_Streak_Length']

    colunas_finais_ordenadas = colunas_estatisticas + colunas_dinamicas_jejum + colunas_resumo_finais

    # Criar DataFrame estruturado
    df_matriz_completa = pd.DataFrame(rows_data, columns=colunas_finais_ordenadas)

    # 5. Salvar em Excel para você analisar perfeitamente sem quebras de linha
    nome_arquivo = "matriz_ausencia_completa.xlsx"
    df_matriz_completa.to_excel(nome_arquivo, index=False)

for index, row in df_ausencia.iterrows():
    for col in colunas_jejum:
        distribuicao_global_jejuns[int(col)] += int(row[col])

total_sequencias = sum(distribuicao_global_jejuns.values())

# P(Jejum = t) -> Distribuição de probabilidade a priori do tamanho do jejum
prob_prior_jejum = {t: (qtd / total_sequencias) for t, qtd in distribuicao_global_jejuns.items()}

# Reconstrói a matriz temporal para a LSTM

gabarito_sorteios = []

for index, row in df_cronologico.iterrows():

    historico_atrasos.append([atraso_atual[d] for d in range(1, 61)])
    gabarito_sorteios.append(sorteados)
    for d in range(1, 61):

        else: atraso_atual[d] += 1

def objective(trial):
    # Parâmetros da LSTM
    window_size = trial.suggest_int('window_size', 10, 18)
    
    # MODIFICAÇÃO 1: Alargado o range mínimo do Grupo de Elite para dar margem à Quadra
    top_n_ia = trial.suggest_int('top_n_ia', 24, 32)
    
    lstm_1 = trial.suggest_int('lstm_1', 64, 128, step=32)
    dropout = trial.suggest_float('dropout', 0.1, 0.3)
    
    # Parâmetro do Ensemble
    peso_lstm = trial.suggest_float('peso_lstm', 0.3, 0.8)
    peso_bayes = 1.0 - peso_lstm
    
    X_all, y_all, gabarito_all = [], [], []
    for i in range(len(dados_norm) - window_size):
        X_all.append(dados_norm[i:(i + window_size), :])
        y_all.append(dados_norm[i + window_size, :])
        gabarito_all.append(gabarito_sorteios[i + window_size])

    X_all, y_all = np.array(X_all), np.array(y_all)
    train_size = int(len(X_all) * 0.8)
    X_train, X_test = X_all[:train_size], X_all[train_size:]
    y_train = y_all[:train_size]
    gabarito_teste = gabarito_all[train_size:]
    
    # Modelo LSTM enxuto para otimização rápida
    model = Sequential([
        LSTM(lstm_1, return_sequences=False, input_shape=(window_size, 60)),
        Dropout(dropout),
        Dense(64, activation='relu'),
        Dense(60, activation='sigmoid')
    ])
    model.compile(optimizer='adam', loss='mean_squared_error')
    model.fit(X_train, y_train, batch_size=64, epochs=10, verbose=0)
    
    # Avaliação retroativa rápida (últimos 80 concursos do teste)
    previsoes_norm = model.predict(X_test[-80:], verbose=0)
    previsoes_lstm = scaler.inverse_transform(previsoes_norm)
    gabarito_recortado = gabarito_teste[-80:]
    
    atrasos_no_teste = matriz_atrasos[train_size + window_size - 1 : -1][-80:]
    
    score_fim = 0
    for i in range(80):
        pred_lstm = list(previsoes_lstm[i])
        atraso_concurso_atual = atrasos_no_teste[i]
        resultado_real = gabarito_recortado[i]
        
        scores_ensemble = []
        for idx in range(60):
            dezena = idx + 1
            score_lstm_norm = 1 / (1 + pred_lstm[idx])
            jejum_futuro = atraso_concurso_atual[idx] + 1
            score_bayes_norm = prob_prior_jejum.get(jejum_futuro, 0.0001)
            
            score_final = (peso_lstm * score_lstm_norm) + (peso_bayes * score_bayes_norm)
            scores_ensemble.append((dezena, score_final))
            
        dezenas_sugeridas = sorted(scores_ensemble, key=lambda x: x[1], reverse=True)
        top_rodada = [item[0] for item in dezenas_sugeridas[:top_n_ia]]
        
        jogos_simulados = []
        tentativas = 0
        
        # MODIFICAÇÃO 3: Subiu para 400 tentativas para dar mais fôlego ao gerador estrito
        while len(jogos_simulados) < 5 and tentativas < 400:
            tentativas += 1
            palpite = sorted(random.sample(top_rodada, 6))
            if jogo_e_valido_estritamente(palpite):
                jogos_simulados.append(palpite)
                
        # MODIFICAÇÃO 2: Sistema de pontos rebalanceado (Terno zerado para o Optuna focar em Quadras)
        for jogo in jogos_simulados:
            acertos = len(set(jogo) & resultado_real)
            if acertos == 3: score_fim += 0        # Ignora o Terno para forçar a rede a subir o nível
            elif acertos == 4: score_fim += 1000   # Peso massivo para a Quadra
            elif acertos == 5: score_fim += 50000  # Peso mestre para a Quina
            
    return score_fim

    index_atual = target_index - 1
    
    for i in range(1, 61):
        combinacoes_restantes = comb(n - i, k - 1)
        if index_atual < combinacoes_restantes:
            jogo.append(i)
            k -= 1
            if k == 0:
                break
        else:
            index_atual -= combinacoes_restantes
    return jogo

def extrair_features_do_indice(jogo, atrasos_atuais):
    """ Extrai a assinatura estocástica que o CatBoost vai avaliar no jogo gerado """
    probs_bayes = [prob_prior_jejum.get(atrasos_atuais[num-1] + 1, 0.0001) for num in jogo]
    return [
        np.mean(probs_bayes),                  # Média de aderência à curva horizontal
        np.std(probs_bayes),                   # Volatilidade Bayes
        sum(1 for num in jogo if num % 2 == 0), # Balanço de pares
        np.mean(jogo),                         # Centro de massa do volante
        np.std(jogo)                           # Dispersão espacial das dezenas
    ]

    tolerancia_busca = trial.suggest_int('tolerancia_busca', 10000, 30000)
    corte_catboost = trial.suggest_float('corte_catboost', 0.3, 0.7)
    
    X_all, y_all, gabarito_all = [], [], []

    X_train, X_test = X_all[:train_size], X_all[train_size:]
    y_train = y_all[:train_size]
    gabarito_teste = gabarito_all[train_size:]
    
    # 3.1 Treinar Regressor LSTM rápido
    model = Sequential([
        LSTM(64, return_sequences=False, input_shape=(window_size, 60)),
        Dropout(0.2),
        Dense(32, activation='relu'),
        Dense(1, activation='linear')
    ])
    model.compile(optimizer='adam', loss='mean_squared_error')
    model.fit(X_train, y_train, batch_size=64, epochs=8, verbose=0)
    
    # 3.2 Treinar CatBoost Regressor rápido para avaliar os desvios de pontuação
    X_cb_train, y_cb_train = [], []
    atrasos_treino_cb = matriz_atrasos[window_size:train_size]

                        feats = np.array([extrair_features_do_indice(palpite, atraso_atual_concurso)])

            elif acertos == 5: score_fim += 50000

print("Carregando dados e calculando probabilidades de Bayes...")
colunas_bolas = ['Bola1', 'Bola2', 'Bola3', 'Bola4', 'Bola5', 'Bola6']
df_cronologico = df_mega_sena.sort_values(by='Concurso').copy()

# Carrega a tabela para extrair a probabilidade condicional de cada tamanho de jejum
df_ausencia = pd.read_excel("matriz_ausencia_completa.xlsx")
colunas_jejum = [col for col in df_ausencia.columns if str(col).isdigit()]

distribuicao_global_jejuns = {int(col): 0 for col in colunas_jejum}
for index, row in df_ausencia.iterrows():
    for col in colunas_jejum:
        distribuicao_global_jejuns[int(col)] += int(row[col])

total_sequencias = sum(distribuicao_global_jejuns.values())

# P(Jejum = t) -> Distribuição de probabilidade a priori do tamanho do jejum
prob_prior_jejum = {t: (qtd / total_sequencias) for t, qtd in distribuicao_global_jejuns.items()}

# Reconstrói a matriz temporal para a LSTM
historico_atrasos = []
gabarito_sorteios = [] 
atraso_atual = {d: 0 for d in range(1, 61)}

for index, row in df_cronologico.iterrows():
    sorteados = set([int(row[c]) for c in colunas_bolas])
    historico_atrasos.append([atraso_atual[d] for d in range(1, 61)])
    gabarito_sorteios.append(sorteados)
    for d in range(1, 61):
        if d in sorteados: atraso_atual[d] = 0
        else: atraso_atual[d] += 1

matriz_atrasos = np.array(historico_atrasos)
scaler = MinMaxScaler(feature_range=(0, 1))
dados_norm = scaler.fit_transform(matriz_atrasos)


# =============================================================
# 2. FUNÇÃO OBJETIVO: OPTUNA TUNANDO O ENSEMBLE COM MIRA NAS QUADRAS
# =============================================================
def objective(trial):
    # Parâmetros da LSTM
    window_size = trial.suggest_int('window_size', 10, 18)
    
    # MODIFICAÇÃO 1: Alargado o range mínimo do Grupo de Elite para dar margem à Quadra
    top_n_ia = trial.suggest_int('top_n_ia', 24, 32)
    
    lstm_1 = trial.suggest_int('lstm_1', 64, 128, step=32)
    dropout = trial.suggest_float('dropout', 0.1, 0.3)
    
    # Parâmetro do Ensemble
    peso_lstm = trial.suggest_float('peso_lstm', 0.3, 0.8)
    peso_bayes = 1.0 - peso_lstm
    
    X_all, y_all, gabarito_all = [], [], []
    for i in range(len(dados_norm) - window_size):
        X_all.append(dados_norm[i:(i + window_size), :])
        y_all.append(dados_norm[i + window_size, :])
        gabarito_all.append(gabarito_sorteios[i + window_size])

    X_all, y_all = np.array(X_all), np.array(y_all)
    train_size = int(len(X_all) * 0.8)
    X_train, X_test = X_all[:train_size], X_all[train_size:]
    y_train = y_all[:train_size]
    gabarito_teste = gabarito_all[train_size:]
    
    # Modelo LSTM enxuto para otimização rápida
    model = Sequential([
        LSTM(lstm_1, return_sequences=False, input_shape=(window_size, 60)),
        Dropout(dropout),
        Dense(64, activation='relu'),
        Dense(60, activation='sigmoid')
    ])
    model.compile(optimizer='adam', loss='mean_squared_error')
    model.fit(X_train, y_train, batch_size=64, epochs=10, verbose=0)
    
    # Avaliação retroativa rápida (últimos 80 concursos do teste)
    previsoes_norm = model.predict(X_test[-80:], verbose=0)
    previsoes_lstm = scaler.inverse_transform(previsoes_norm)
    gabarito_recortado = gabarito_teste[-80:]
    
    atrasos_no_teste = matriz_atrasos[train_size + window_size - 1 : -1][-80:]
    
    score_fim = 0
    for i in range(80):
        pred_lstm = list(previsoes_lstm[i])
        atraso_concurso_atual = atrasos_no_teste[i]
        resultado_real = gabarito_recortado[i]
        
        scores_ensemble = []
        for idx in range(60):
            dezena = idx + 1
            score_lstm_norm = 1 / (1 + pred_lstm[idx])
            jejum_futuro = atraso_concurso_atual[idx] + 1
            score_bayes_norm = prob_prior_jejum.get(jejum_futuro, 0.0001)
            
            score_final = (peso_lstm * score_lstm_norm) + (peso_bayes * score_bayes_norm)
            scores_ensemble.append((dezena, score_final))
            
        dezenas_sugeridas = sorted(scores_ensemble, key=lambda x: x[1], reverse=True)
        top_rodada = [item[0] for item in dezenas_sugeridas[:top_n_ia]]
        
        jogos_simulados = []
        tentativas = 0
        
        # MODIFICAÇÃO 3: Subiu para 400 tentativas para dar mais fôlego ao gerador estrito
        while len(jogos_simulados) < 5 and tentativas < 400:
            tentativas += 1
            palpite = sorted(random.sample(top_rodada, 6))
            if jogo_e_valido_estritamente(palpite):
                jogos_simulados.append(palpite)
                
        # MODIFICAÇÃO 2: Sistema de pontos rebalanceado (Terno zerado para o Optuna focar em Quadras)
        for jogo in jogos_simulados:
            acertos = len(set(jogo) & resultado_real)
            if acertos == 3: score_fim += 0        # Ignora o Terno para forçar a rede a subir o nível
            elif acertos == 4: score_fim += 1000   # Peso massivo para a Quadra
            elif acertos == 5: score_fim += 50000  # Peso mestre para a Quina
            
    return score_fim

# =============================================================
# 3. FUNÇÃO CORRIGIDA PARA CONVERTER ÍNDICE EM 6 NÚMEROS
# =============================================================

def indice_para_combinacao(index, n=60, k=6):
    """
    Converte um índice (base 1) de volta para os 6 números da Mega-Sena.
    Garante que o índice esteja dentro do limite antes de começar.
    """
    total_combinacoes = comb(n, k)
    
    # Proteção de limites contra pequenas variações numéricas da rede
    if index < 1:
        index = 1
    elif index > total_combinacoes:
        index = total_combinacoes

    combinacao = []
    current = 1
    
    for i in range(k):
        elementos_restantes = k - 1 - i
        
        while current <= n:
            c = comb(n - current, elementos_restantes)
            
            if index <= c:
                combinacao.append(current)
                current += 1  # O próximo número precisa ser maior que o atual
                break
            
            index -= c
            current += 1
            
    return combinacao


# Converter o índice previsto em jogo real
jogo_previsto = indice_para_combinacao(indice_previsto)

print("\n" + "-" * 50)
